    % Init node
    \node (start) [draw, ellipse] {Start};

    \node at (bigbox.north) [below=0.1cm, xshift=-3.8cm] {\textit{Input}};

    \node (systemfile) [draw, trapezium, trapezium left angle=75, trapezium right angle=105, below of=start, node distance=2.3cm, xshift=-3.1cm] {System file};

    \node (excitonfile) [draw, trapezium, trapezium left angle=75, trapezium right angle=105, below of=start, node distance=2.3cm, xshift=3.0cm] {Exciton file};

    % Screening file node
    \node (screeningfile) [draw, trapezium, trapezium left angle=75, trapezium right angle=105, below of=start, node distance=2.3cm, xshift=0.0cm] {Screening file};

    %%% Main algorithm
    % Diagonalize H0 node
    \node (computeH0) [draw, rectangle, below of=bigbox, node distance=2.5cm, align=center]
    {Diagonalize $H_0(\mathbf{k})\ \forall \mathbf{k}\in$ BZ \\ Store $\{\varepsilon_{n\mathbf{k}}\}$ and $\{\bC^{n \bk}\}$};

    % Method decision node
    \node (method) [draw, diamond, aspect=2, below of=computeH0, node distance=2.5cm, align=center] {Method to use};

    \node (auxRS) [left=of method, xshift=-1.5cm] {};

    % Auxiliar node for Reciprocal space method
    \node (auxRecS) [right=of method, xshift=1.5cm] {};

    \node (screening) [draw, diamond, aspect=2, below of=auxRecS, node distance=2.0cm, align=center] {Screening file?};

    \node (computeRS) [draw, rectangle, below of=auxRS, node distance=2.0cm, align=center] {Compute and \\store $V_{ij}(\mathbf{k})$};

    \node (computeepsilon) [draw, rectangle, below of=method, node distance=2.0cm, align=center] {Compute and \\store $\varepsilon^{-1}_{\mathbf{G}\mathbf{G}'}(\mathbf{q})$};

    % Compute BSE node
    \node (computeBSE) [draw, rectangle, below of=computeepsilon, node distance=2.0cm, align=center]
    {Compute BSE \\ $H_{vc,v'c'}(\mathbf{k},\mathbf{k}',\mathbf{Q})$};

    % Diagonalize BSE node
    \node (diagonalize) [draw, rectangle, below of=computeBSE, node distance=2.cm, align=center]
    {Diagonalize BSE \\ Store $E_X^n(\mathbf{Q})$ and $A^n_{vc}(\mathbf{k},\mathbf{Q})$};

    \node (converge_energies) [draw, diamond, aspect=2, below of=diagonalize, node distance=2.5cm, align=center] {$E^n_X(\mathbf{Q})$ converged?};

    %%% Output
    % Box containing output files
    \node (bigbox_output) [draw, rectangle, minimum width=10cm, minimum height=2cm, below of=converge_energies, node distance=3cm] {};

    \node at (bigbox_output.north) [below=0.1cm, xshift=-4cm] {\textit{Output}};

    \node (wavefunctions) [draw, trapezium, trapezium left angle=75, trapezium right angle=105, below of=bigbox_output, node distance=0.3cm, xshift=-2.3cm, align=center]
    {Wavefunctions\\ $|\psi^n_X(\mathbf{k})|^2$, $|\psi^n_X(\mathbf{r}_e,\mathbf{r}_h)|^2$};

    \node (optical_conductivity) [draw, trapezium, trapezium left angle=75, trapezium right angle=105, below of=bigbox_output, node distance=0.3cm, xshift=2.3cm, align=center]

    \node (increaseNk) [draw, rectangle, right of=converge_energies, node distance=6.3cm, align=center] {Increase $N_{\mathbf{k}}$ \\ and/or $N_{v/c}$ \\ and/or $G_c$/$G^{X}_c$};

    \node (End) [draw, ellipse, below of=symmetry, node distance=2.8cm] {End};

    \draw [->] (bigbox_output) -- (symmetry);
    \draw [->] (symmetry) -- node[anchor=east] {Yes} (End);
    \draw [->] (symmetry) -- node[anchor=south, xshift=-1cm] {No} ++(6.3,0) -- (increaseNk);
