\section{Static Noninteracting Polarizability and Time-Reversal Symmetry}
\label{app:polarizability_TRS_proof}

In this section we derive expression~\eqref{eq:static_polarizability_TRS} in Section~\ref{subsec:non_interacting_polarizability} from its more general form~\eqref{eq:static_polarizability_repeated}, assuming the system is time-reversal symmetric.
In a periodic system with no contribution to the Hamiltonian that violates \gls{trs}, there exists a unitary transformation $T$ such that

\begin{equation}
    T H^*(-\bk) T^{\dagger} = H(\bk) \,
\end{equation}

\noindent is verified for the Bloch Hamiltonian $H(\bk)$.
Apart from systems with \gls{soc}, our Hamiltonian is diagonal in the spin.
If there is spin then the transformation of \gls{trs} is sightly more complicated, but we refrain from using it, and will just ignore spin.
Either way, we have verified numerically that for models where \gls{soc} is present, considering it or ignoring it makes no relevant difference in the calculation of the polarizability.
Also, for time-reversal invariant systems, the quasiparticle energies naturally obey $\epsilon_{n  \bk} = \epsilon_{n - \bk}$.

The expression~\eqref{eq:static_polarizability_repeated} in the main text for the polarizability reads

\begin{equation}
\begin{aligned}
     \chi^0_{\bG\bG'}(\bq) = & \frac{1}{\area} \sum_{\vv \cc,\bk \sigma} \Bigg[\frac{\langle \cc,\bk|\ee^{-\ii(\bq+\bG)\vdot\br}|\vv,\bk+\bq\rangle \langle \vv,\bk+\bq|\ee^{\ii(\bq+\bG')\vdot\br}|\cc,\bk\rangle}{\epsilon_{\vv \bk+\bq} - \epsilon_{\cc \bk}} \Bigg] + \\
                             & \frac{1}{\area} \sum_{\vv \cc,\bk \sigma} \Bigg[\frac{\langle \vv,\bk|\ee^{-\ii(\bq+\bG)\vdot\br}|\cc,\bk+\bq\rangle \langle \cc,\bk+\bq|\ee^{\ii(\bq+\bG')\vdot\br}|\vv,\bk\rangle}{\epsilon_{\vv \bk} - \epsilon_{\cc \bk+\bq}}\Bigg] \,.
\end{aligned}
\end{equation}

\noindent Now we show that the second term on the right-hand side of the previous expression equals the first one.
First, we recall that the sum over momentum $\bk$ is over momenta in the \gls{bz}, which is center symmetric.
Therefore, we can replace $\bk \to -\bk$ and then shift $\bk \to \bk + \bq$ to obtain

\begin{equation}
\begin{aligned}
    &\sum_{\vv \cc,\bk\sigma} \frac{\langle \vv,-\bk-\bq|\ee^{-\ii(\bq+\bG)\vdot\br}|\cc,-\bk\rangle \langle \cc,-\bk|\ee^{\ii(\bq+\bG')\vdot\br}|\vv,-\bk-\bq\rangle}{\epsilon_{\vv -\bk-\bq} - \epsilon_{\cc -\bk}} = \\
    & \sum_{\vv \cc,\bk\sigma} \frac{\langle \vv,-\bk-\bq|\ee^{-\ii(\bq+\bG)\vdot\br}|\cc,-\bk\rangle \langle \cc,-\bk|\ee^{\ii(\bq+\bG')\vdot\br}|\vv,-\bk-\bq\rangle}{\epsilon_{\vv \bk+\bq} - \epsilon_{\cc \bk}} \,
\end{aligned}
\end{equation}

\noindent where we used \gls{trs} to write the equality, namely $\epsilon_{n  \bk} = \epsilon_{n - \bk}$.
Now we only need some sort of symmetry for the plane-wave matrix elements of the form $\langle n,\bk|\ee^{-\ii\bq\vdot\br}|n',\bk'\rangle$.
For that, we recall here their expression in terms of the eigenvectors of the Bloch Hamiltonian, that we denote by $\bC^{n \bk}$.
Being eigenvectors of the Bloch Hamiltonian, the tight-binding coefficients obey

\begin{equation}
\begin{gathered}
    H(\bk) \bC^{n \bk} = \epsilon_{n \bk} \bC^{n \bk} \Leftrightarrow H(-\bk) \bC^{n -\bk} = \epsilon_{n -\bk} \bC^{n -\bk} \Leftrightarrow \\
    \Leftrightarrow  H^*(-\bk) (\bC^{n -\bk})^* = \epsilon_{n \bk} (\bC^{n -\bk})^*  \Leftrightarrow T H^*(-\bk) T^{\dagger} T (\bC^{n -\bk})^* = \epsilon_{n -\bk} T(\bC^{n -\bk})^* \Leftrightarrow \\
    \Leftrightarrow  H(\bk) T (\bC^{n -\bk})^* = \epsilon_{n \bk} T(\bC^{n -\bk}) \,,
\end{gathered} 
\end{equation}

\noindent where, to get the last equality, we used the identities verified by $H(\bk)$ and $\epsilon_{n \bk}$ introduced above.
So, in principle, we have $T (\bC^{n -\bk})^* = \bC^{n \bk}$.
Now, for the matrix elements of interest, we introduce the following notation:

\begin{equation}
     \langle n,\bk | \ee^{-\ii(\bk' - \bk + \bG) \vdot \br} | n',\bk' \rangle = \sum_{i \alpha} (C_{i \alpha}^{n \bk})^* C_{i \alpha}^{n' \bk'}  \ee^{-\ii(\bk' - \bk + \bG)  \vdot \bt_i} \equiv (\bC^{n \bk})^{\dagger} \vdot \big(\bC^{n' \bk'} \odot \underline{\varphi} \big) \,
\end{equation}

\noindent where the vector $\underline{\varphi}$ has by entries the phases $\ee^{-\ii(\bk' - \bk + \bG)  \vdot \bt_i}$, that is, $(\underline{\varphi})_{i\alpha} = \ee^{-\ii(\bk' - \bk + \bG)  \vdot \bt_i}$, and $\odot$ is used to denote element-wise product, usually called Hadamard product in the literature.
Then, we only need to employ the identity $T (\bC^{n -\bk})^* = \bC^{n \bk}$ to obtain

\begin{align}
    \langle n,\bk | \ee^{-\ii(\bk' - \bk + \bG) \vdot \br} | n',\bk' \rangle & =   (\bC^{n \bk})^{\dagger} \vdot \big(\bC^{n' \bk'} \odot \underline{\varphi} \big) =  \Big((\bC^{n -\bk})^{\mathrm{T}} T^{\dagger} \Big) \vdot \Big( \big(T (\bC^{n' -\bk'})^* \big) \odot \underline{\varphi} \Big) = \notag \\
    & =  (\bC^{n -\bk})^{\mathrm{T}} (T^{\dagger}  T) \Big((\bC^{n' -\bk'})^* \odot \underline{\varphi}\Big) = (\bC^{n' -\bk'})^{\dagger} \vdot \big(\bC^{n -\bk} \odot \underline{\varphi}\big) = \notag \\
    & =  \langle n',-\bk' | \ee^{-\ii(\bk' - \bk + \bG) \vdot \br} | n,-\bk \rangle \,,
\end{align}

\noindent with $\mathrm{T}$ denoting transpose.
In this way, the second term in the polarizability can be rewritten as

\begin{equation}
\begin{aligned}
    & \sum_{\vv \cc,\bk \sigma} \frac{\langle \vv,-\bk-\bq|\ee^{-\ii(\bq+\bG)\vdot\br}|\cc,-\bk\rangle \langle \cc,-\bk|\ee^{\ii(\bq+\bG')\vdot\br}|\vv,-\bk-\bq\rangle}{\epsilon_{\vv \bk+\bq} - \epsilon_{\cc \bk}} = \\
    & \sum_{\vv \cc,\bk \sigma} \frac{\langle \cc,\bk |\ee^{-\ii(\bq+\bG)\vdot\br}|\vv,\bk+\bq\rangle \langle \vv,\bk+\bq|\ee^{\ii(\bq+\bG')\vdot\br}|\cc,\bk\rangle}{\epsilon_{\vv \bk+\bq} - \epsilon_{\cc \bk}}\,,
\end{aligned}
\end{equation}

\noindent which concludes our proof.
This identity has also been verified numerically.

\section{Regularization of the dielectric function}
\label{app:regularization}

In this appendix, we explain in detail how the divergences of the dielectric matrix in the long-wavelength limit $\bq \to \bzero$ are treated in this work.
As commonly done in the literature, different matrix elements of the dielectric matrix have different treatment, namely the head $\varepsilon_{\bzero \bzero}$, the wing $\varepsilon_{\bG \bzero}$ and $\varepsilon_{\bzero \bG}$, and body $\varepsilon_{\bG\bG'}$ matrix elements, with $\bG,\bG' \neq \bzero$.
As discussed in the main text, we adopt the symmetrized dielectric matrix defined by

\begin{equation}
\label{eq:symmetrized_dielectric_matrix}
\varepsilon_{\bG \bG'} (\bq) = \delta_{\bG \bG'} - \sqrt{v_c(\bq + \bG)} \chi^0_{\bG \bG'} (\bq) \sqrt{v_c(\bq + \bG')} \,,
\end{equation}

\noindent which eases the technical issues related to the long-wavelength limit, as we explain next.
We have already seen, in Subsection~\ref{subsec:dielectric_matrix_inverse}, that for the head element it does not matter whether we symmetrize the dielectric function or not.
This is because

\begin{align}
    \varepsilon_{\bzero \bzero} (\bq \to \bzero) & =  1 - \sqrt{v_c(\bq \to \bzero)} \chi^0_{\bzero \bzero} (\bq \to \bzero) \sqrt{v_c(\bq \to \bzero)} = 1 - v_c(\bq \to \bzero) \chi^0_{\bzero \bzero} (\bq \to \bzero) = \notag \\
    & = 1 - \frac{e}{2 \varepsilon_0 q} \bq \vdot \mathbf{B} \bq \to 1 \,,
\end{align}

\noindent where the last equality holds for very small $\bq$, and $\mathbf{B}$ is the Hessian matrix of the function $\chi^0_{\bzero \bzero} (q_x,q_y)$ at the point $(q_x,q_y) = (0,0)$.
The Hessian, here a 2 by 2 matrix, does not depend on the small vector $\bq$, because the function $\chi^0_{\bzero \bzero} (\bq)$ is analytic in $\bq$ around the origin according to Pick, Cohen and Martin in~\cite{Pick_Cohen_Martin_1970}.
In the same paper, they explain that all the functions $\chi^0_{\bG \bG'} (\bq)$, for any $\bG$ and $\bG'$ vectors, are analytic at $\bq = \bzero$.
In other words, the functions $\chi^0_{\bG \bG'} (\bq)$ can be expanded around the origin in powers of $\bq$.
For our analysis, we only need to expand it up to second order in Taylor series.
Therefore, we can write for the wing terms

% \begin{align}
%     \varepsilon_{\bG \bzero} (\bq \to \bzero) & =  - \sqrt{v_c(\bG)} \chi^0_{\bG \bzero} (\bq \to \bzero) \sqrt{v_c(\bq \to \bzero)} = - \sqrt{v_c(\bG)} \frac{e}{2 \varepsilon_0 \sqrt{q}} \bq \vdot \mathbf{A}(\bG) \,\, \to 0 \text{ with } \mathcal{O} (\sqrt{q}) \,, \\
%     \varepsilon_{\bzero \bG} (\bq \to \bzero) & =  - \sqrt{v_c(\bq \to \bzero)} \chi^0_{\bG \bzero} (\bq \to \bzero) \sqrt{v_c(\bG)} =  - \sqrt{v_c(\bG)} \frac{e}{2 \varepsilon_0 \sqrt{q}} \bq \vdot \mathbf{A}^*(\bG) \to 0 \text{ with } \mathcal{O} (\sqrt{q}) \,,
% \end{align}

\begin{align}
    \varepsilon_{\bG \bzero} (\bq \to \bzero) & = -\sqrt{v_c(\bG)} \chi^0_{\bG \bzero} (\bq \to \bzero) \sqrt{v_c(\bq \to \bzero)} = \notag \\
                                              & = -\frac{e}{2 \varepsilon_0 \sqrt{|\bG|} \sqrt{q}} \bq \vdot \mathbf{A}(\bG) \,\,\, \propto \sqrt{q} \to 0 \,, \\
    \varepsilon_{\bzero \bG} (\bq \to \bzero) & = -\sqrt{v_c(\bq \to \bzero)} \chi^0_{\bzero \bG} (\bq \to \bzero) \sqrt{v_c(\bG)} = \notag \\
                                              & = -\frac{e}{2 \varepsilon_0 \sqrt{|\bG|} \sqrt{q}} \bq \vdot \mathbf{A}(\bG)^{*} \propto \sqrt{q} \to 0 \,,
\end{align}

\noindent where $\mathbf{A}(\bG)$ is the Jacobian matrix (a vector, actually) of the function $\chi^0_{\bG \bzero} (\bq)$ at $\bq = \bzero$, which does not depend on $\bq$ due to $\chi^0_{\bG \bzero} (\bq)$ being analytic, as mentioned above.

If we were to use the nonsymmetric dielectric matrix, then the wing terms $\varepsilon_{\bG \bzero}$ would also tend to zero as $\bq \to \bzero$, but with $q$ instead, but the terms $\varepsilon_{\bzero \bG}$ would tend to $\varepsilon_{\bzero \bG}(\bq \to \bzero) \to -e \hat{\bq} \vdot \mathbf{A}(\bG)^{*}/2\varepsilon_0$, which is finite and depends on the direction along which one takes the limit $\bq \to \bzero$.
One can see the usefulness of using the symmetrized dielectric matrix, making it easier to implement numerically.
OWe just set $\varepsilon_{\bzero \bG}(\bzero) = \varepsilon_{\bG \bzero}(\bzero) = 0$, leaving an eventual long-wavelength regularization for later when dealing with the screened potential.

Now, we investigate the behaviour of the inverse dielectric matrix for near-zero $\bq$, in the light of Refs.~\cite{Rasmussen_Schmidt_Winther_Thygesen_2016} and~\cite{Pick_Cohen_Martin_1970}, by writing the dielectric matrix as a matrix by blocks.
We separate the head, the wing, and the body terms, which we represent by $\varepsilon_{\bzero \bzero}$, $\varepsilon_{\bG \bzero}$ and $\varepsilon_{\bzero \bG}$, and $B_{\bG \bG'}$, respectively.
Skipping the details of the inversion, we write directly

\begin{align}
    \varepsilon^{-1}_{\bzero \bzero}       & =  \left[\varepsilon_{\bzero \bzero} - \sum_{\bG,\bG' \neq \bzero} \varepsilon_{\bzero \bG} B^{-1}_{\bG \bG'} \varepsilon_{\bG'\bzero}\right]^{-1} \,, \\
    \varepsilon^{-1}_{\bG \bzero}     & =  -\varepsilon^{-1}_{\bzero \bzero} \sum_{\bG' \neq \bzero}  B^{-1}_{\bG \bG'}  \varepsilon_{\bG' \bzero}\,, \\
    \varepsilon^{-1}_{\bzero \bG}     & =  -\varepsilon^{-1}_{\bzero \bzero} \sum_{\bG' \neq \bzero}  \varepsilon_{\bzero\bG'} B^{-1}_{\bG' \bG} \,, \\
    \varepsilon^{-1}_{\bG \bG'} & =  B^{-1}_{\bG \bG'} + \varepsilon^{-1}_{\bzero \bzero} \left(\sum_{\bG'' \neq \bzero}  B^{-1}_{\bG \bG''}  \varepsilon_{\bG'' \bzero}\right) \left(\sum_{\bG'' \neq \bzero} \varepsilon_{\bzero\bG''} B^{-1}_{\bG'' \bG'}\right) \,.
\end{align}

\noindent Note that in the previous relations, $B^{-1}_{\bG \bG'}$ designates the $(\bG,\bG')$ matrix element of the inverse of the body, and not the inverse of the body element $B_{\bG \bG'}$.
Given the properties of the wing matrix elements derived above when taking the long-wavelength limit, we can infer that

\begin{align}
     \varepsilon^{-1}_{\bzero \bzero} (\bq \to \bzero) &\to 1 \,, \\
     \varepsilon^{-1}_{\bG \bG'} (\bq \to \bzero) &\to B^{-1}_{\bG \bG'}(\bq \to \bzero) \,,
\end{align}

\noindent as one could expect for the head element of a strictly \bd~dielectric matrix, while the body of the inverse dielectric matrix is simply the inverse of the body of the dielectric matrix.
However, the analysis of the wing elements needs more care, since we need to know how they behave for small $\bq$.
Using the very same properties of the wing elements $\varepsilon_{\bG \bzero}$ and $\varepsilon_{\bzero \bG}$, we can write

\begin{align}
    &\varepsilon^{-1}_{\bG \bzero} (\bq \to \bzero) \to  \sum_{\bG' \neq \bzero}  B^{-1}_{\bG \bG'}   \frac{e}{2 \varepsilon_0 \sqrt{\bG'}} \sqrt{q} \hat{\bq} \vdot \mathbf{A}(\bG') \,\,\to 0 \,,\\
    &\varepsilon^{-1}_{\bzero \bG} (\bq \to \bzero) \to  \sum_{\bG' \neq \bzero}  B^{-1}_{\bG' \bG}  \frac{e}{2 \varepsilon_0 \sqrt{\bG'}} \sqrt{q} \hat{\bq} \vdot \mathbf{A}(\bG')^{*} \to 0 \,,
\end{align}

\noindent so the wing terms of the inverse dielectric matrix tend to zero when $\bq$ approaches zero, just as expected for a \bd~system~\cite{Guandalini_Varsano_2023}.
We are in position to address the troublesome matrix elements of the screened potential at zero momentum.

We know now that $\varepsilon^{-1}_{\bzero \bzero} (\bq \to \bzero) \to 1$, but the bare Coulomb potential diverges with $1/q$, hence, the head term of the screened potential $W_{\bzero \bzero} (\bq)$ diverges with $1/q$ in the long-wavelength limit.
Nonetheless, the $\bq = \bzero$ term in reciprocal space calculations is known to be of upmost relevance to make calculations converge faster~\cite{Rasmussen_Schmidt_Winther_Thygesen_2016}.
Fortunately, we know that $\varepsilon^{-1}_{\bzero \bzero} (\bq) \to 1$ has a linear behavior for small $\bq$.
Therefore, we can use an analytical expression to regularize the term $W_{\bzero \bzero} (\bzero)$, as suggested by F.~Hüser, T.~Olsen and K.S.~Thygesen in Ref.~\cite{Huser_Olsen_Thygesen_2013}.
Their suggestion is approximating the divergent term by

\begin{equation}
    W_{\bzero \bzero} (\bzero) = \frac{1}{\area_{\Gamma}} \int_{\area_{\Gamma}} \mathrm{d} \bq \, v_c(\bq) \big[1 + \bq \vdot \mathbf{\nabla}_{\bq} \varepsilon^{-1}_{\bzero \bzero} (\bq)|_{\bq = \bzero} \big] \,,
\end{equation}

\noindent where $\area_{\Gamma}$ is a small circular region around the origin of the \gls{bz}, the point $\Gamma$.
This integral has an analytical expression.
For an anisotropic material, we can simply take $\varepsilon^{-1}_{\bzero \bzero} (\bq)|_{\bq = \bzero} = (-r_{0,x},-r_{0,y})$, where the parameters $r_{0,x}$ and $r_{0,y}$ are the screening parameters computed numerically along the $x$ and $y$ directions, respectively.
The expression evaluates to

\begin{equation}
    \label{eq:W00_regularization}
    W_{\bzero \bzero} (\bzero) = \frac{e}{\varepsilon_0 q_0} \left[1 - \frac{1}{4} q_0 (r_{0,x}+r_{0,y})\right] =  \frac{e}{\varepsilon_0 q_0} \left(1 - \frac{q_0 r_0}{2} \right) \,,
\end{equation}

\noindent where the second equality holds for isotropic materials, and $q_0$ is the radius of $\area_{\Gamma}$.
Typically, for $q_0$ we choose a percentage of the magnitude of the $\bk$ point in the \bz~mesh closest to the origin, which depends on the number of $\bk$ points used, and of course the geometry of the \bz\@.
The exact value for the percentage may also differ from material to material, but we have it as an input parameter for Xatu.

Regarding the wing terms of the screened potential, using the symmetrized dielectric function we get

\begin{align}
    W_{\bG \bzero} (\bq \to 0) & =  \sqrt{v_c(\bG)} \varepsilon^{-1}_{\bG \bzero} (\bq \to \bzero) \sqrt{v_c (\bq \to \bzero)} = \notag \\
                               & = -\frac{e}{2\varepsilon_0\sqrt{|\bG|}} \hat{\bq} \vdot \sum_{\bG' \neq \bzero} \frac{1}{ \sqrt{\bG'}} \mathbf{A}(\bG') B^{-1}_{\bG' \bG}\,, \\
    W_{\bzero \bG} (\bq \to 0) & =  \sqrt{v_c (\bq \to \bzero)} \varepsilon^{-1}_{\bzero \bG} (\bq \to \bzero) \sqrt{v_c(\bG)}  =  \notag \\
                               & = -\frac{e}{2\varepsilon_0\sqrt{|\bG|}} \hat{\bq} \vdot \sum_{\bG' \neq \bzero} \frac{1}{ \sqrt{\bG'}} \mathbf{A}(\bG')^* B^{-1}_{\bG \bG'}\,, 
\end{align}

\noindent with both results depending on the direction of the vector $\bq$ as it approaches zero.
Even though they do not tend to zero, they can be averaged around the origin over a center-symmetric region, such that they can be deemed null.
This is due to the fact that the dependence on the direction of the vector $\bq$ is linear in its versor $\hat{\bq} = \bq/|\bq|$.
This contrasts with the case in Rasmussen et al.'s work~\cite{Rasmussen_Schmidt_Winther_Thygesen_2016}, where the angular dependence of the wing terms of the screened potential is not trivial, in such a way that the average has to be evaluated numerically.
Therein, for the head term the dependence on $\hat{\bq}$ is also not trivial, but the simplicity of the expression allows for an approximate analytical evaluation.
Our expression for the regularization of $W_{\bzero \bzero}(\bzero)$ is analytical, as shown in Eq.~\eqref{eq:W00_regularization}

\section{Quasi-2D}

In this section we derive an expression for the dielectric function,
in a representation that is a mix between reciprocal space and real
space, for in-plane and off-plane directions, respectively. We start
from it's general expression in real space

\begin{equation}
\varepsilon(\br,\br') = \delta(\br-\br')-\int v_{c}(\br-\br'')\chi^{0}(\br'',\br')\dr''\,,
\end{equation}

\noindent where we have omitted the frequency dependence, as we focus only on static screening.
We want to work with the dielectric function in terms of momentum, so we compute

\begin{equation}
\begin{aligned}
    \varepsilon(\bq,\bq',z,z') &  = \int \dd \boldsymbol{\rho} \int \dd \boldsymbol{\rho}'\ee^{\ii\bq \vdot\boldsymbol{\rho}}\ee^{-\ii\bq'\vdot\boldsymbol{\rho}'}\varepsilon(\br,\br')\\
    \varepsilon_{\bG \bG'}(\bq,z,z') &  = \delta_{\bG \bG'} \delta( z-z')-\int \dd z''\frac{ \ee^{-|\bq +\bG||z-z''|}}{2\epsz|\bq +\bG|}\chi^{0}_{\bG \bG'}(\bq,z'',z')\,,
\end{aligned}
\end{equation}

\noindent where to obtain the final result we replaced the Coulomb potential
with the inverse 2D Fourier transform of its 3D real space representation
(as the bare Coulomb potential is invariant under continuous translations,
we only need one momentum variable), given by
\begin{equation}
v_{c}(\br-\br') = \frac{1}{\sqrt{\area}}\sum_{\bk} \ee^{\ii\bk\vdot\boldsymbol{\rho}}v_{c}( \bk,z-z') = \frac{1}{\sqrt{\area}}\sum_{\bk} \ee^{\ii\bk\vdot\boldsymbol{\rho}}\frac{e}{2\epsz k} \ee^{-k|z-z'|}\,,\,k = |\bk|\,,
\end{equation}

\noindent where $\area$ is the area occupied by the 2D material, and the polarizability $\chi^{0}$ by it's mixed representation

\begin{equation}
\chi^{0}(\br,\br') = \frac{1}{\area}\sum_{\bk,\bk'} \ee^{\ii\bk\vdot\boldsymbol{\rho}} \ee^{-\ii\bk'\vdot\boldsymbol{\rho}'} \chi^{0}(\bk,\bk';z,z')\,,
\end{equation} 

\noindent in concomitance with the fact that $\chi^{0}(\bq,\bq',z,z')\neq0$
if and only if $\bq -\bq'\neq \bG$
for any reciprocal lattice vector $\bG$, thus defining
$\chi^{0}_{\bG \bG'}(\bq,z,z') = \chi^{0}( \bq +\bG,\bq +\bG',z,z')$,
and used the identity
\begin{equation}
\int \dd \boldsymbol{\rho} \, \ee^{\ii( \bk-\bk') \vdot\boldsymbol{\rho}} = \sqrt{\area} \delta_{\bk,\bk'}
\end{equation}

\noindent Now we just have to sort out the object $\chi^{0}_{\bG \bG'}(\bq,z,z')$,
which is actually treated by Thygesen et. al in ``Dielectric Genome
of van der Waals Heterostructures``.

\subsection{Static irreducible polarizability within RPA}

Let us start from its static non-interacting expression obtained from
the expression above:
\begin{equation}
\chi^{0}(\br,\br') = \sum_{\vv\bk,c'\bk'}\left[\frac{\psi_{\cc,\bk'}(\br)\psi_{\vv,\bk}^{*}(\br)\psi_{\vv,\bk}(\br')\psi_{\cc,\bk'}^{*}(\br')}{(\epsilon_{\vv\bk}-\epsilon_{\cc\bk'})+ \ii \alpha}-\frac{\psi_{\cc,\bk'}^{*}(\br)\psi_{\vv,\bk}(\br)\psi_{\vv,\bk}^{*}(\br')\psi_{\cc,\bk'}(\br')}{(\epsilon_{\cc\bk'}-\epsilon_{\vv\bk})+ \ii \alpha}\right]\,.
\end{equation}

Replacing each Bloch state wavefunction by their expression in the
TB approximation, that we recall is
\begin{equation}
\psi_{n,\bk}(\br) = \frac{1}{\sqrt{N}}\sum_{\bR} \ee^{\ii\bk\vdot\bR}\sum_{i\alpha}C_{i\alpha}^{n\bk}\phi_{\alpha}(\br - \bR - \bt_i)\,,
\end{equation}

\noindent we get
\begin{equation}
\begin{aligned}\chi^{0}(\br,\br') &  = \sum_{\vv\bk,c'\bk'}\left[\frac{\psi_{\cc,\bk'}(\br)\psi_{\vv,\bk}^{*}(\br)\psi_{\vv,\bk}(\br')\psi_{\cc,\bk'}^{*}(\br')}{(\epsilon_{\vv\bk}-\epsilon_{\cc\bk'})+ \ii \alpha}-\frac{\psi_{\cc,\bk'}^{*}(\br)\psi_{\vv,\bk}(\br)\psi_{\vv,\bk}^{*}(\br')\psi_{\cc,\bk'}(\br')}{(\epsilon_{\cc\bk'}-\epsilon_{\vv\bk})+ \ii \alpha}\right] = \\

 & \sum_{j\beta}\left(C_{j\beta}^{\cc\bk+\bq }\right)^{*}C_{j\beta}^{\vv\bk} \ee^{-\ii (\bq +\bG) \vdot \bt_{j}}\sum_{i\alpha}\left(C_{i\alpha}^{\vv\bk}\right)^{*}C_{i\alpha}^{\cc\bk+\bq } \ee^{\ii (\bq +\bG') \vdot\bt_{i}}\Bigg\}\,.
\end{aligned}
\end{equation}

In this way, one can identify in the previous expression the strictly
2D dielectric function we have been looking at and implemented in
the code, through the identity
\begin{equation}
\varepsilon_{\bG \bG'}(\bq )\equiv \lim_{d \to 0} \frac{\bar{\varepsilon}_{\bG \bG'}(\bq )}{d}\,.
\end{equation}

If we take instead the limit $|\bq +\bG|\to\infty$,
we see that the $d$-dependence becomes irrelevant. So we expect that
the averaged dielectric function does not differ much from the strictly
2D one for materials of small thickness,such as monolayer hBN, in
the high momentum limit (to confirm numerically, also for MoS2).

\section{Inverse dielectric function within RPA}

In this section, we work out an expression for the inverse of the
dielectric function, necessary for the screened Coulomb potential
in order to compute the interaction matrix elements in the BSH. According
to Martin's textbook
\begin{equation}
\varepsilon^{-1} ( \br,\br';\omega) = \delta(\br-\br')+\int v_{c}( \br-\br'')\chi( \br'',\br;\omega)\dr''\,,
\end{equation}

where $\chi$ is the interacting reducible polarizability, that relates
to the irreducible one through the Dyson's equation
\begin{equation}
\chi = \chi^{\irr}+\chi^{\irr}*v_{c}*\chi\,,
\end{equation}

where the symbol $*$ denotes the operation of convolution. Computing
the full polarizability at all interacting levels is too much of an
endeavor, so approximations are necessary. First for the irreducible
polarizability, instead of considering all the irreducible diagrams
in the expansion, one keeps only the bubble or Hartree diagram. As
a consequence, within RPA we have
\begin{equation}
\chi^{\RPA} = \chi^{0}+\chi^{0}*v_{c}*\chi^{\RPA}\,,
\end{equation}

where the full interacting polarizability is approximated to an interacting
one at the single-particle level. Making now explicit the momentum
and $z$ coordinate dependence on the previous expression we write
\begin{equation}
\chi^{\RPA}_{\bG \bG'}(\bq,z,z') = \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}} \int \int \,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1}) v_{c}( \bq +\bG_{1},z_{1}-z_{2}) \chi^{\RPA}_{\bG_{1} \bG'}(\bq,z_{2},z')\dd z_{1}\dd z_{2}
\end{equation}

Let us assume that we can determine $\chi^{\RPA}_{\bG \bG'}(\bq,z,z')$.
Then, for the inverse dielectric function within RPA we would have
\begin{equation}
\varepsilon_{\bG \bG'}^{-1}(\bq,z,z') = \delta_{\bG \bG'} \delta(z-z')+\int v_{c}( \bq +\bG,z-z'')\chi^{\RPA}_{\bG \bG'}(\bq,z'',z')\dd z''\,.
\end{equation}

In principle, assuming we have the inverse of the dielectric function,
we can compute the interaction matrix elements. We outline their computation
in the next equation

\begin{equation}
\begin{aligned} & D_{vc,v'c'}( \bk,\bk',\bQ) = \int\dr\int\dr'\,\psi_{c,\bk + \bQ}^{*}(\br)\psi_{v',\bk'}^{*}( \br')W( \br,\br')\psi_{c',\bk'+\bQ}(\br)\psi_{v,\bk}( \br') = \\
 &  = \int\dd z\int\dd z'\int \dd \boldsymbol{\rho} \int \dd \boldsymbol{\rho}'\psi_{c,\bk + \bQ}^{*}(\br)\psi_{v',\bk'}^{*}( \br')( \sum_{\bq }^{\BZ}\sum_{\bG \bG'} \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}W_{\bG \bG'}(\bq,z,z') \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'})\psi_{c',\bk'+\bQ}(\br)\psi_{v,\bk}( \br') = \\
 &  = \sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\int\psi_{c,\bk + \bQ}^{*}(\br) \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}\psi_{c',\bk'+\bQ}(\br)\dd\boldsymbol{\rho}\int\psi_{v',\bk'}^{*}( \br') \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}\psi_{v,\bk}( \br')\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \int( \sum_{\bR_{1}} \ee^{-\ii ( \bk+\bQ) \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}\phi_{\alpha}^{*}(\br - \bR_{1} - \bt_i )) \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}( \sum_{\bR_{2}} \ee^{\ii ( \bk'+\bQ) \vdot\bR_{2}}\sum_{i'\alpha'}C_{i'\alpha'}^{c'\bk'+\bQ}\phi_{\alpha'}(\br - \bR_{2} - \bt_{i'} ))\dd\boldsymbol{\rho}\\
 & \int( \sum_{\bR_{3}} \ee^{-\ii\bk'\vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}\phi_{\beta}^{*}(\br' - \bR_{3} - \bt_{j} )) \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}( \sum_{\bR_{4}} \ee^{\ii\bk\vdot\bR_{4}}\sum_{j'\beta'}C_{j'\beta'}^{\vv\bk}\phi_{\beta'} (\br' - \bR_{4} - \bt_{j'} ))\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \int\sum_{\bR_{1},\bR_{2}} \ee^{-\ii ( \bk+\bQ) \vdot\bR_{1}} \ee^{\ii ( \bk'+\bQ) \vdot\bR_{2}}\sum_{i\alpha,i'\alpha'}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i'\alpha'}^{c'\bk'+\bQ} \delta( \boldsymbol{\rho}-\bR_{1}-\bt_{i}^{\parallel})\delta( z-t_{i}^{z})\delta_{i\alpha,i'\alpha'} \delta_{\bR_{1},\bR_{2}}\ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}\dd\boldsymbol{\rho}\\
 & \int\sum_{\bR_{3},\bR_{4}} \ee^{-\ii\bk'\vdot\bR_{3}} \ee^{\ii\bk\vdot\bR_{4}}\sum_{j\beta,j'\beta'}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j'\beta'}^{\vv\bk} \delta( \boldsymbol{\rho}'-\bR_{3}-\bt_{j}^{\parallel})\delta( z'-t_{j}^{z})\delta_{j\beta,j'\beta''} \delta_{\bR_{3},\bR_{4}}\ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \sum_{\bR_{1}} \ee^{-\ii ( \bk-\bk') \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot ( \bR_{1}+\bt_{i})} \delta( z-t_{i}^{z})\\
 & \sum_{\bR_{3}} \ee^{-\ii ( \bk'-\bk) \vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot(\bR_{3}+\bt_{j} )} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \sum_{\bR_{1}} \ee^{-\ii ( \bk-\bk'-\bq ) \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\\
 & \sum_{\bR_{3}} \ee^{-\ii ( \bk'-\bk+\bq ) \vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')N\delta_{\bq ,\bk-\bk'}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\int\int W_{\bG \bG'}( \bk-\bk',z,z')\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\sum_{j\beta}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z}) = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\sum_{j\beta}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{ \ii ( \bk-\bk') \vdot ( \bt_{i}-\bt_{j})} \ee^{ \ii \bG\vdot \bt_{i}} \ee^{- \ii \bG'\vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z}) = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z})\,,
\end{aligned}
\end{equation}

where the point-like orbital approximation has been employed, and
the last two lines are completely equivalent expressions, but the
last one might be more appropriate for implementation purposes. To
evaluate said expression numerically, we need to compute $W_{\bG \bG'}( \bk,t_{i}^{z},t_{j}^{z})$
for every $\bk\in\BZ$. For that, we already have
from before the inverse of the dielectric function. At any pair of
points $( z,z')$, we can use the expression


\begin{equation}
W_{\bG \bG'}( \bk,z,z') = \int\varepsilon_{\bG \bG'}^{-1}(\bq,z,z'')v_{c}( \bq +\bG',z''-z')\dd z''\,,
\end{equation}

where $\varepsilon_{\bG \bG'}^{-1}(\bq,z,z')$
is the inverse of the dielectric matrix in the $( q,z)$
mixed representation within RPA, which was previously determined.
Replacing it by its expression above one arrives at
\begin{equation}
\begin{aligned}W_{\bG \bG'}(\bq,z,z') =  & \int\left[\delta_{\bG \bG'} \delta(z-z'')+\int v_{c}( \bq +\bG,z-z_{1})^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')\dd z_{1}\right]v_{c}( \bq +\bG',z''-z')\dd z'' = \\
 & \delta_{\bG \bG'}v_{c}( \bq +\bG',z-z')+\int\int v_{c}( \bq +\bG,z-z_{1})\chi^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-z')\dd z_{1}\dd z''\,.
\end{aligned}
\end{equation}

As in the point-like orbital approximation we only need to compute
the screened potential between $z$ points located at the heights
of the atoms in the motif, we can work out the previous expression
at those points, to be used for the interaction matrix elements as
\begin{equation}
W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) = \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\int v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-t_{j}^{z})\dd z_{1}\dd z''.
\end{equation}

However, in practice one can not solve Dyson's equation for $\chi^{\RPA}{\bG \bG'}(\bq,z_{1},z'')$
directly, as we model the $z$-dependence in the polarizability
$\chi^{0}_{\bG \bG'}(\bq,z,z')$
with Dirac delta functions. One solution is to working with a basis
of potential functions $\left\{ \phi_{\alpha}( z)\right\} $,
that assume the form $\phi_{\alpha}( z)\sim( z-z_{0})^{\alpha}$,
and compute the respective matrix elements of the polarizability in
that basis (which is what Thygesen does in its Dielectric Genome paper).
As a workaround, one can for instance develop an iterative process,
in which the interaction matrix elements are computed with a self-consistently
determined RPA polarizability. To do this, let us first recall Dyson's
equation for the polarizability within RPA
\begin{equation}
\chi^{\RPA}_{\bG \bG'}(\bq,z,z') = \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \int \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{\RPA}_{\bG_{1} \bG'}(\bq,z_{2},z')\,,
\end{equation}

for which an iterative process looks like
\begin{equation}
\begin{aligned}
\text{Step 0:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')\\
\text{Step 1:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z')\\
\text{Step 2:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z')+\\
 & \sum_{\bG_{1},\bG_{2}} \int \dd z_{1} \dd z_{2} \dd z_{3} \dd z_{4}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG_{2}}(\bq,z_{2,}z_{3})\times\\
 & v_{c}( \bq +\bG_{2},z_{3}-z_{4})\chi^{0}_{\bG_{2},\bG'}(\bq,z_{4},z')\\
 & \vdots
\end{aligned}
\end{equation}

Along this line of thought, we can evaluate numerically the screened
potential in the mixed representation with a self-consistent method
as:

\begin{equation}
\begin{aligned}
\text{Step 0:}\,W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\dd z \dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z)\chi^{0}_{\bG \bG'}(\bq,z,z'')v_{c}( \bq +\bG',z''-t_{j}^{z})\\
\text{Step 1:}\,W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\dd z \dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z)\chi^{0}_{\bG \bG'}(\bq,z,z'')v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1}} \int \dd z \dd z'' \dd z_{1} \dd z_{2} \,v_{c}( \bq +\bG,t_{i}^{z}-z)\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z'')\times\\
 & v_{c}( \bq +\bG',z''-t_{j}^{z})\\
\text{Step 2:}\, W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int \dd z_{1}\dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{0}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1}} \int\dd z \dd z'' \dd z_{1} \dd z_{2}\,v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{0}_{\bG,\bG_{1}}(\bq,z_{1},s_{1})v_{c}( \bq +\bG_{1},s_{1}-s_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,s_{2},z'')\\
 & \times v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1},\bG_{2}} \int\dd z \dd z'' \dd z_{1} \dd z_{2} \dd z_{3} \dd z_{4}\,v_{c}( \bq +\bG,t_{i}^{z}-z)\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\times\\
 & \chi^{0}_{\bG_{1} \bG_{2}}(\bq,z_{2},z_{3}) v_{c}( \bq +\bG_{2},z_{3}-z_{4})\chi^{0}_{\bG_{2},\bG'}(\bq,z_{4},z'')v_{c}( \bq +\bG',z''-t_{j}^{z}) \\
 & \vdots
\end{aligned}
\end{equation}
