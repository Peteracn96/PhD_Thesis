\label{app:Q2D_multilayer}

\section{Quasi-2D polarizability and dielectric function}

In this section we recall the polarizability and the dielectric function in the mixed representation from the main text.

\begin{align}
    \chi^{0}_{\bG \bG'}(\bq,z,z') =  &\frac{1}{\area}\sum_{\vv \cc,\bk \sigma} \frac{1}{( \epsilon_{\vv\bk}-\epsilon_{\cc\bk+\bq })}\Bigg\{ \\
    &\sum_{i\alpha}\Big[C_{i\alpha}^{\cc\bk+\bq }\left(C_{i\alpha}^{\vv\bk}\right)^{*} \ee^{-\ii (\bq +\bG) \vdot\bt_{i}} \delta(z - t_{i,z}) \Big] \sum_{j\beta}\Big[C_{j\beta}^{\vv\bk}\left(C_{j\beta}^{\cc\bk+\bq }\right)^{*} \ee^{\ii (\bq +\bG') \vdot \bt_{j}} \delta(z' - t_{j,z}) \Big] + \notag\\
    &\sum_{i\alpha}\Big[\left(C_{i\alpha}^{\cc\bk+\bq }\right)^{*}C_{i\alpha}^{\vv\bk}\, \ee^{-\ii (\bq +\bG) \vdot\bt_{i}} \delta(z - t_{i,z}) \Big] \sum_{j\beta} \Big[\left(C_{j\beta}^{\vv\bk}\right)^{*}C_{j\beta}^{\cc\bk+\bq }\, \ee^{\ii (\bq +\bG') \vdot \bt_{j}} \delta(z' - t_{j,z})\Big]\Bigg\}\,. \notag
\end{align}

and 

\begin{align}
    \label{eq:quasi_2D_dielectric_function_final}
    \varepsilon_{\bG \bG'}(\bq,z,z') = & \,\delta_{\bG \bG'} \delta( z-z')-\frac{1}{2\epsz |\bq +\bG|\area}\sum_{\vv \cc, \bk \sigma}\frac{1}{( \epsilon_{\vv\bk}-\epsilon_{\cc\bk+\bq })}\Bigg\{\\ 
    & \sum_{i\alpha} \Big[C_{i\alpha}^{\cc \bk+\bq} \left(C_{i\alpha}^{\vv\bk}\right)^{*} \ee^{-\ii (\bq +\bG) \vdot\bt_{i}} \ee^{-|\bq +\bG||z-t_{i,z}|}\Big] \sum_{j\beta}\Big[C_{j\beta}^{\vv\bk}\left(C_{j\beta}^{\cc\bk+\bq }\right)^{*} \ee^{\ii (\bq +\bG') \vdot \bt_{j}} \delta(z' - t_{j,z})\Big]+  \notag\\
    & \sum_{i\alpha} \Big[\left(C_{i\alpha}^{\cc\bk+\bq }\right)^{*}C_{i\alpha}^{\vv\bk} \ee^{-\ii (\bq +\bG) \vdot\bt_{i}} \ee^{-|\bq +\bG||z-t_{i,z}|}\Big] \sum_{j\beta}\Big[\left(C_{j\beta}^{\vv\bk}\right)^{*}C_{j\beta}^{\cc\bk+\bq } \ee^{\ii (\bq +\bG') \vdot \bt_{j}} \delta(z' - t_{j,z})\Big]\Bigg\} \,. \notag
\end{align}


\section{RPA inverse dielectric function}

In this section, we work out an expression for the inverse of the
dielectric function, necessary for the screened Coulomb potential
in order to compute the interaction matrix elements in the BSH. According
to Martin's textbook

\begin{equation}
\varepsilon^{-1} (\br,\br';\omega) = \delta(\br - \br')+\int v_{c}(\br - \br'')\chi(\br'',\br;\omega)\dr''\,,
\end{equation}

where $\chi$ is the interacting reducible polarizability, that relates to the irreducible one through the Dyson's equation

\begin{equation}
\chi = \chi^{\irr}+\chi^{\irr}*v_{c}*\chi\,,
\end{equation}

where the symbol $*$ denotes the operation of convolution. Computing
the full polarizability at all interacting levels is too much of an
endeavor, so approximations are necessary. First for the irreducible
polarizability, instead of considering all the irreducible diagrams
in the expansion, one keeps only the bubble or Hartree diagram. As
a consequence, within \rpa~we have

\begin{equation}
\chi^{\RPA} = \chi^{0}+\chi^{0}*v_{c}*\chi^{\RPA}\,,
\end{equation}

where the full interacting polarizability is approximated to an interacting
one at the single-particle level. Making now explicit the momentum
and $z$ coordinate dependence on the previous expression we write
\begin{equation}
\chi^{\RPA}_{\bG \bG'}(\bq,z,z') = \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}} \int \int \,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1}) v_{c}( \bq +\bG_{1},z_{1}-z_{2}) \chi^{\RPA}_{\bG_{1} \bG'}(\bq,z_{2},z')\dd z_{1}\dd z_{2}
\end{equation}

Let us assume that we can determine $\chi^{\RPA}_{\bG \bG'}(\bq,z,z')$.
Then, for the inverse dielectric function within \rpa~we would have
\begin{equation}
\varepsilon_{\bG \bG'}^{-1}(\bq,z,z') = \delta_{\bG \bG'} \delta(z-z')+\int v_{c}( \bq +\bG,z-z'')\chi^{\RPA}_{\bG \bG'}(\bq,z'',z')\dd z''\,.
\end{equation}

In principle, assuming we have the inverse of the dielectric function,
we can compute the interaction matrix elements. We outline their computation
in the next equation

\begin{equation}
\begin{aligned} & D_{vc,v'c'}( \bk,\bk',\bQ) = \int\dr\int\dr'\,\psi_{c,\bk + \bQ}^{*}(\br)\psi_{v',\bk'}^{*}(\br')W(\br,\br')\psi_{c',\bk'+\bQ}(\br)\psi_{v,\bk}(\br') = \\
 &  = \int\dd z\int\dd z'\int \dd \boldsymbol{\rho} \int \dd \boldsymbol{\rho}'\psi_{c,\bk + \bQ}^{*}(\br)\psi_{v',\bk'}^{*}(\br')( \sum_{\bq }^{\BZ}\sum_{\bG \bG'} \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}W_{\bG \bG'}(\bq,z,z') \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'})\psi_{c',\bk'+\bQ}(\br)\psi_{v,\bk}(\br') = \\
 &  = \sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\int\psi_{c,\bk + \bQ}^{*}(\br) \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}\psi_{c',\bk'+\bQ}(\br)\dd\boldsymbol{\rho}\int\psi_{v',\bk'}^{*}(\br') \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}\psi_{v,\bk}(\br')\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \int( \sum_{\bR_{1}} \ee^{-\ii ( \bk+\bQ) \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}\phi_{\alpha}^{*}(\br - \bR_{1} - \bt_i )) \ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}( \sum_{\bR_{2}} \ee^{\ii ( \bk'+\bQ) \vdot\bR_{2}}\sum_{i'\alpha'}C_{i'\alpha'}^{c'\bk'+\bQ}\phi_{\alpha'}(\br - \bR_{2} - \bt_{i'} ))\dd\boldsymbol{\rho}\\
 & \int( \sum_{\bR_{3}} \ee^{-\ii\bk'\vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}\phi_{\beta}^{*}(\br' - \bR_{3} - \bt_{j} )) \ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}( \sum_{\bR_{4}} \ee^{\ii\bk\vdot\bR_{4}}\sum_{j'\beta'}C_{j'\beta'}^{\vv\bk}\phi_{\beta'} (\br' - \bR_{4} - \bt_{j'} ))\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \int\sum_{\bR_{1},\bR_{2}} \ee^{-\ii ( \bk+\bQ) \vdot\bR_{1}} \ee^{\ii ( \bk'+\bQ) \vdot\bR_{2}}\sum_{i\alpha,i'\alpha'}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i'\alpha'}^{c'\bk'+\bQ} \delta( \boldsymbol{\rho}-\bR_{1}-\bt_{i}^{\parallel})\delta( z-t_{i}^{z})\delta_{i\alpha,i'\alpha'} \delta_{\bR_{1},\bR_{2}}\ee^{ \ii ( \bq +\bG) \vdot\boldsymbol{\rho}}\dd\boldsymbol{\rho}\\
 & \int\sum_{\bR_{3},\bR_{4}} \ee^{-\ii\bk'\vdot\bR_{3}} \ee^{\ii\bk\vdot\bR_{4}}\sum_{j\beta,j'\beta'}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j'\beta'}^{\vv\bk} \delta( \boldsymbol{\rho}'-\bR_{3}-\bt_{j}^{\parallel})\delta( z'-t_{j}^{z})\delta_{j\beta,j'\beta''} \delta_{\bR_{3},\bR_{4}}\ee^{- \ii ( \bq +\bG') \vdot\boldsymbol{\rho}'}\dd\boldsymbol{\rho}'\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \sum_{\bR_{1}} \ee^{-\ii ( \bk-\bk') \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot ( \bR_{1}+\bt_{i})} \delta( z-t_{i}^{z})\\
 & \sum_{\bR_{3}} \ee^{-\ii ( \bk'-\bk) \vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot(\bR_{3}+\bt_{j} )} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')\\
 & \sum_{\bR_{1}} \ee^{-\ii ( \bk-\bk'-\bq ) \vdot\bR_{1}}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\\
 & \sum_{\bR_{3}} \ee^{-\ii ( \bk'-\bk+\bq ) \vdot\bR_{3}}\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N^{2}}\sum_{\bq }^{\BZ}\sum_{\bG \bG'}\int\int W_{\bG \bG'}(\bq,z,z')N\delta_{\bq ,\bk-\bk'}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bq +\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bq +\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\int\int W_{\bG \bG'}( \bk-\bk',z,z')\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \delta( z-t_{i}^{z})\sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}} \delta( z'-t_{j}^{z})\dd z\,\dd z' = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\sum_{j\beta}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z}) = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\sum_{j\beta}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{ \ii ( \bk-\bk') \vdot ( \bt_{i}-\bt_{j})} \ee^{ \ii \bG\vdot \bt_{i}} \ee^{- \ii \bG'\vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z}) = \\
 &  = \frac{1}{N}\sum_{\bG \bG'}\sum_{i\alpha}\left(C_{i\alpha}^{\cc\bk\boldsymbol{+}\bQ}\right)^{*}C_{i\alpha}^{c'\bk'+\bQ}\ee^{ \ii ( \bk-\bk'+\bG) \vdot \bt_{i}} \sum_{j\beta}\left(C_{j\beta}^{v'\bk'}\right)^{*}C_{j\beta}^{\vv\bk}\ee^{- \ii ( \bk-\bk'+\bG') \vdot \bt_{j}}W_{\bG \bG'}( \bk-\bk',t_{i}^{z},t_{j}^{z})\,,
\end{aligned}
\end{equation}

where the point-like orbital approximation has been employed, and
the last two lines are completely equivalent expressions, but the
last one might be more appropriate for implementation purposes. To
evaluate said expression numerically, we need to compute $W_{\bG \bG'}( \bk,t_{i}^{z},t_{j}^{z})$
for every $\bk\in\BZ$. For that, we already have
from before the inverse of the dielectric function. At any pair of
points $( z,z')$, we can use the expression


\begin{equation}
W_{\bG \bG'}( \bk,z,z') = \int\varepsilon_{\bG \bG'}^{-1}(\bq,z,z'')v_{c}( \bq +\bG',z''-z')\dd z''\,,
\end{equation}

where $\varepsilon_{\bG \bG'}^{-1}(\bq,z,z')$
is the inverse of the dielectric matrix in the $( q,z)$
mixed representation within \rpa~, which was previously determined.
Replacing it by its expression above one arrives at
\begin{equation}
\begin{aligned}W_{\bG \bG'}(\bq,z,z') =  & \int\left[\delta_{\bG \bG'} \delta(z-z'')+\int v_{c}( \bq +\bG,z-z_{1})^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')\dd z_{1}\right]v_{c}( \bq +\bG',z''-z')\dd z'' = \\
 & \delta_{\bG \bG'}v_{c}( \bq +\bG',z-z')+\int\int v_{c}( \bq +\bG,z-z_{1})\chi^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-z')\dd z_{1}\dd z''\,.
\end{aligned}
\end{equation}

As in the point-like orbital approximation we only need to compute
the screened potential between $z$ points located at the heights
of the atoms in the motif, we can work out the previous expression
at those points, to be used for the interaction matrix elements as
\begin{equation}
W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) = \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\int v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{\RPA}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-t_{j}^{z})\dd z_{1}\dd z''.
\end{equation}

However, in practice one can not solve Dyson's equation for $\chi^{\RPA}{\bG \bG'}(\bq,z_{1},z'')$
directly, as we model the $z$-dependence in the polarizability
$\chi^{0}_{\bG \bG'}(\bq,z,z')$
with Dirac delta functions. One solution is to working with a basis
of potential functions $\left\{ \phi_{\alpha}( z)\right\} $,
that assume the form $\phi_{\alpha}( z)\sim( z-z_{0})^{\alpha}$,
and compute the respective matrix elements of the polarizability in
that basis (which is what Thygesen does in its Dielectric Genome paper).
As a workaround, one can for instance develop an iterative process,
in which the interaction matrix elements are computed with a self-consistently
determined \rpa~polarizability. To do this, let us first recall Dyson's
equation for the polarizability within \rpa~
\begin{equation}
\chi^{\RPA}_{\bG \bG'}(\bq,z,z') = \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \int \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{\RPA}_{\bG_{1} \bG'}(\bq,z_{2},z')\,,
\end{equation}

for which an iterative process looks like
\begin{equation}
\begin{aligned}
\text{Step 0:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')\\
\text{Step 1:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z')\\
\text{Step 2:}\,\chi^{\RPA}_{\bG \bG'}(\bq,z,z') =  & \chi^{0}_{\bG \bG'}(\bq,z,z')+\sum_{\bG_{1}}\int\dd z_{1} \dd z_{2}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z')+\\
 & \sum_{\bG_{1},\bG_{2}} \int \dd z_{1} \dd z_{2} \dd z_{3} \dd z_{4}\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG_{2}}(\bq,z_{2,}z_{3})\times\\
 & v_{c}( \bq +\bG_{2},z_{3}-z_{4})\chi^{0}_{\bG_{2},\bG'}(\bq,z_{4},z')\\
 & \vdots
\end{aligned}
\end{equation}

Along this line of thought, we can evaluate numerically the screened
potential in the mixed representation with a self-consistent method
as:

\begin{equation}
\begin{aligned}
\text{Step 0:}\,W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\dd z \dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z)\chi^{0}_{\bG \bG'}(\bq,z,z'')v_{c}( \bq +\bG',z''-t_{j}^{z})\\
\text{Step 1:}\,W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int\dd z \dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z)\chi^{0}_{\bG \bG'}(\bq,z,z'')v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1}} \int \dd z \dd z'' \dd z_{1} \dd z_{2} \,v_{c}( \bq +\bG,t_{i}^{z}-z)\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,z_{2},z'')\times\\
 & v_{c}( \bq +\bG',z''-t_{j}^{z})\\
\text{Step 2:}\, W_{\bG \bG'}(\bq,t_{i}^{z},t_{j}^{z}) =  & \delta_{\bG \bG'}v_{c}( \bq +\bG,t_{i}^{z}-t_{j}^{z})+\int \dd z_{1}\dd z'' v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{0}_{\bG \bG'}(\bq,z_{1},z'')v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1}} \int\dd z \dd z'' \dd z_{1} \dd z_{2}\,v_{c}( \bq +\bG,t_{i}^{z}-z_{1})\chi^{0}_{\bG,\bG_{1}}(\bq,z_{1},s_{1})v_{c}( \bq +\bG_{1},s_{1}-s_{2})\chi^{0}_{\bG_{1} \bG'}(\bq,s_{2},z'')\\
 & \times v_{c}( \bq +\bG',z''-t_{j}^{z})+\\
 & \sum_{\bG_{1},\bG_{2}} \int\dd z \dd z'' \dd z_{1} \dd z_{2} \dd z_{3} \dd z_{4}\,v_{c}( \bq +\bG,t_{i}^{z}-z)\,\chi^{0}_{\bG,\bG_{1}}(\bq,z,z_{1})v_{c}( \bq +\bG_{1},z_{1}-z_{2})\times\\
 & \chi^{0}_{\bG_{1} \bG_{2}}(\bq,z_{2},z_{3}) v_{c}( \bq +\bG_{2},z_{3}-z_{4})\chi^{0}_{\bG_{2},\bG'}(\bq,z_{4},z'')v_{c}( \bq +\bG',z''-t_{j}^{z}) \\
 & \vdots
\end{aligned}
\end{equation}
